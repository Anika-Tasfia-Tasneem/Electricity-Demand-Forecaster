var DAYS=['Sat','Sun','Mon','Tue','Wed','Thu','Fri'];
var seed=7;function rnd(){seed=(seed*16807)%2147483647;return seed/2147483647}
function gauss(){return (rnd()+rnd()+rnd()+rnd()-2)*1.7}
var data=[],T=0,t0=new Date(),playing=true,speed=2,H=24,model='adapt',tempAdj=0,prog=1,hover=-1;

function gen(n){
  seed=Math.floor(Math.random()*9999)+1;var d=[];
  for(var i=0;i<n;i++){
    var h=i%24,dow=Math.floor(i/24)%7;
    var shape=.82+.1*Math.exp(-Math.pow(h-13,2)/20)+.34*Math.exp(-Math.pow(h-20,2)/7)-.1*Math.exp(-Math.pow(h-4,2)/8);
    var wk=dow===6?.9:(dow===5?.97:1);
    d.push(Math.round(11200*shape*wk*(1+gauss()*.012)));
  }
  return d;
}
function stamp(i){var s=new Date(t0.getTime()+(i-T)*36e5);return s}
function fmtH(i){var s=stamp(i);return DAYS[(i%168<0?0:Math.floor(i/24)%7)]+' '+('0'+s.getHours()).slice(-2)+':00'}
function hourOf(i){return i%24}

/* forecasting: uses only values at or before origin */
function level(origin){
  if(model!=='adapt')return 1;var a=0,b=0;
  for(var k=0;k<24;k++){var i=origin-k,p=base(i,origin-24,true);if(p){a+=data[i];b+=p}}
  return b?Math.max(.8,Math.min(1.2,a/b)):1;
}
function base(idx,origin,raw){
  if(model==='naive'&&!raw){var j=idx-168;return j<=origin&&j>=0?data[j]:null}
  var s=0,c=0;for(var k=1;k<=4;k++){var j2=idx-168*k;if(j2<=origin&&j2>=0){s+=data[j2];c++}}
  return c?s/c:null;
}
function predict(idx,origin){
  var b=base(idx,origin);if(b===null)return null;
  return b*(model==='adapt'?level(origin):1);
}
function metrics(){
  var e=0,n=0,sq=0;
  for(var i=T-23;i<=T;i++){var p=predict(i,i-1);if(p){var r=(data[i]-p);e+=Math.abs(r)/data[i];sq+=r*r;n++}}
  return {mape:n?e/n*100:0,sd:n?Math.sqrt(sq/n):300};
}
function forecast(){
  var m=metrics(),f=[],adj=1+.018*tempAdj;
  for(var h=1;h<=H;h++){
    var p=predict(T+h,T);if(p===null)p=data[T];
    p*=adj;var w=m.sd*1.28*Math.sqrt(1+h/12);
    f.push({i:T+h,v:p,lo:p-w,hi:p+w});
  }
  return {f:f,m:m};
}

}

function addFeed(){

function tick(){
  if(!playing)return;
  var i=T+1,h=i%24,dow=Math.floor(i/24)%7;
  var shape=.82+.1*Math.exp(-Math.pow(h-13,2)/20)+.34*Math.exp(-Math.pow(h-20,2)/7)-.1*Math.exp(-Math.pow(h-4,2)/8);
  var wk=dow===6?.9:(dow===5?.97:1);
  var surge=Math.sin(i/40)*.03;
  data[i]=Math.round(11200*shape*wk*(1+surge+gauss()*.012));T=i;
  addFeed();refresh(false);
}
